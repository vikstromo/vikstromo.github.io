Today I added drums to the module. 4 sounds that are randomly chosen at each trigger. It sounds terrible without fx, but quite nice with some echo and blur. 

I also added a routine that makes a little better choices for switching presets. It use my slow-listener instead of just making random choices or using current values. So if I play soft for a very long time, it can decide to move to a more intense mood. 

````
SynthDef(\drums, {|out = 0, vol = 0.6, threshold = 0.01, chaos = 0.5, dens = 0.5, space = 0.5, focus = 0.5, ampBus, pitchBus, centBus, flatBus, atk = 0.01, rel = 0.1, gate = 1,
	noiseFloor = 0.0003, softAmp = 0.003, atkThresh = 0.01|

	var amp, micTrig, selfTrig, resetSignal;
	var selector, tempo, sig, mainEnv, effectiveAmp;
	var burstGate, burstClock, burstCount, triggerBurst, numHits;
	var kick, snare, cymbal, riser;
	var autoClock, timeSinceLastMic, waitTime, isAutoMode, autoVol;
	var kickDrift, riserRes, snareTuning;

	amp = In.kr(ampBus);
	amp = Amplitude.kr(amp, 0.01, 0.2);
	micTrig = Trig1.kr(amp > atkThresh, 0.05);

	// triggers 
	timeSinceLastMic = Sweep.kr(micTrig, 1);
	waitTime = TRand.kr(dens.linlin(0,1, 1, 3), dens.linlin(0,1, 3, 12), micTrig);
	isAutoMode = (timeSinceLastMic > waitTime) * (chaos > 0.7);

	autoClock = Impulse.kr(4);
	selfTrig = autoClock * isAutoMode;
	selfTrig = Trig1.kr(selfTrig, 0.05);

	resetSignal = (micTrig + selfTrig).clip(0, 1);

	// To create sequences
	numHits = dens.linexp(0.001, 1, 2, 64);
	tempo = dens.linlin(0, 1, 4, 35).max(0.1); // Safety
	burstClock = Impulse.kr(tempo * LFDNoise1.kr(tempo).range(1, 1 + (chaos * 2)).max(0.1));

	burstCount = PulseCount.kr(burstClock, resetSignal);
	burstGate = SetResetFF.kr(resetSignal, burstCount >= numHits);
	triggerBurst = burstClock * burstGate;

	// Adjust volume to playing, or not if it is auto-playing
	autoVol = isAutoMode.lag(1.0);
	effectiveAmp = Latch.kr(Select.kr(micTrig, [softAmp * 4 * autoVol, amp]), resetSignal);

	selector = Demand.kr(triggerBurst, resetSignal, Dwhite(0, 3));

	// --- MICRO-RANDOMNESS (Added .max to prevent 0-value crashes) ---
	kickDrift = Demand.kr(triggerBurst, 0, Dwhite(-2.0, 2.0));
	snareTuning = Demand.kr(triggerBurst, 0, Dwhite(0.9, 1.1)).max(0.1);
	riserRes = Demand.kr(triggerBurst, 0, Dwhite(0.1, 0.5)).max(0.01);

	// 5. ENGINES
	kick = {
		var t = triggerBurst * (selector absdif: 0 < 0.1);
		var e = EnvGen.ar(Env.perc(0.001, 0.2 + (space * 0.3)), t);
		var pEnv = EnvGen.ar(Env.perc(0.001, 0.03), t) * 120;
		var baseFreq = (48 + kickDrift + (chaos * 15)).max(20);

		var osc = SinOsc.ar(baseFreq + pEnv);
		var click = BPF.ar(Hasher.ar(Sweep.ar(t)), 2000, 0.4) * EnvGen.ar(Env.perc(0.001, 0.01), t);

		((osc + (click * 2)).tanh) * e * effectiveAmp.linlin(noiseFloor, atkThresh, 0.5, 1);
	}.value;

	snare = {
		var t = triggerBurst * (selector absdif: 1 < 0.1);
		var sig;
		var e = EnvGen.ar(Env.perc(0.001, 0.15 + (space * 0.2)), t);
		var snap = BPF.ar(Hasher.ar(Sweep.ar(t)), (2500 * snareTuning).max(100), 0.6);
		sig = ((SinOsc.ar((180 * snareTuning).max(40)) * 0.5) + (snap * 0.8)) * e * effectiveAmp.linlin(noiseFloor, atkThresh, 0.4, 1);
		sig = LPF.ar(sig, TRand.kr(triggerBurst, 500, 6000));
	}.value;

	cymbal = {
		var t = triggerBurst * (selector absdif: 2 < 0.1);
		var e = EnvGen.ar(Env.perc(0.01, 0.3 + (2 * space)), t);
		var sig;
		var metal = BPF.ar(Hasher.ar(Sweep.ar(t)), [6500, 8200, 9400, 12000], 0.05).sum;
		sig = HPF.ar(metal * 3, 4000) * e * effectiveAmp.linlin(noiseFloor, atkThresh, 0.4, 1);
		sig = LPF.ar(sig, LFPulse.kr(0.25).range(2000, 6000));

	}.value;

	riser = {
		var t = triggerBurst * (selector absdif: 3 < 0.1);
		var e = EnvGen.ar(Env.perc(0.1 + (space * 2), 0.05 + (1 - focus * 0.2)), t);
		var sig;
		var sweep = e.linlin(0, 1, 100, chaos.linlin(0, 1, 2000, 12000)).max(50);
		sig = BPF.ar(Hasher.ar(Sweep.ar(t)), sweep, (riserRes + (1 - focus)).max(0.01)) * e * effectiveAmp.linlin(noiseFloor, atkThresh, 0.4, 1);
		sig = LPF.ar(sig, SinOsc.kr(0.5).range(500, 3000));
	}.value;

	sig = (kick + snare + cymbal + riser);
	sig = LeakDC.ar(sig.tanh);
	mainEnv = EnvGen.kr(Env.asr(atk, 1, rel), gate, doneAction: 2);

	Out.ar(out, (sig * vol) * mainEnv ! 2);
}).add;
````
